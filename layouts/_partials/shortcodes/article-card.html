{{- /* Shared card used by the `article` and `list` shortcodes. Both run while Hugo is already rendering page content (search-data.json renders every page), so this partial must never reach anything on the target page that is derived from its rendered .Content - .Content itself, .Summary, .WordCount, .ReadingTime, .TableOfContents. Touching any of them makes Hugo render that page, and two pages whose cards point at each other then deadlock the build with no error output at all. Date, reading time and summary are resolved from front matter and .RawContent instead. A card pointing at the page it sits on renders nothing. @param {page} page The page to render. @param {string} self RelPermalink of the page the card sits on. @param {bool} showSummary Show the page summary. Default true. @param {bool} compactSummary Clamp the summary to a single line. Default false. @param {bool} cover Show the page's cover image when it has one. Default true. @example {{ partial "shortcodes/article-card.html" (dict "page" $p "self" $self) }} */ -}} {{- $showSummary := ne (printf "%v" .showSummary) "false" -}} {{- $compactSummary := eq (printf "%v" .compactSummary) "true" -}} {{- $showCover := ne (printf "%v" .cover) "false" -}} {{- $self := .self -}} {{- with .page -}} {{- if ne .RelPermalink $self -}} {{- $cover := dict -}} {{- if $showCover -}} {{- $cover = partial "utils/page-image.html" (dict "page" . "process" "Resize 400x webp q80") -}} {{- end -}} {{- $date := "" -}} {{- if not .Date.IsZero -}} {{- $date = .Date | time.Format (site.Params.dateFormat | default ":date_long") -}} {{- end -}} {{- /* .RawContent is the file as written, so reading it costs nothing and cannot recurse. Shortcode calls are dropped first: they are markup, not prose, and would otherwise be counted and shown verbatim. */ -}} {{- $raw := replaceRE `(?s)\{\{[%<].*?[%>]\}\}` " " .RawContent -}} {{- /* Hugo's own reading time is ceil(words / 213). */ -}} {{- $readingTime := "" -}} {{- $words := countwords $raw -}} {{- if gt $words 0 -}} {{- $minutes := int (math.Ceil (div (float $words) 213.0)) -}} {{- $readingTime = printf ((T "readingTime") | default "%d min read") $minutes -}} {{- end -}} {{- $summary := "" -}} {{- if $showSummary -}} {{- $summary = .Description | plainify | htmlUnescape -}} {{- if not $summary -}} {{- /* Enough Markdown stripped to read as prose; the card clamps it anyway. */ -}} {{- $text := $raw -}} {{- $text = replaceRE "(?s)```.*?```" " " $text -}} {{- $text = replaceRE `(?m)^\s{0,3}(#{1,6}|>|[-*+]|\d+\.)\s+` "" $text -}} {{- $text = replaceRE `!?\[([^\]]*)\]\([^)]*\)` "$1" $text -}} {{- $text = replaceRE "[*_~`]" "" $text -}} {{- $text = replaceRE `\s+` " " $text -}} {{- $summary = $text | plainify | htmlUnescape | chomp | truncate 200 -}} {{- end -}} {{- end -}} {{- with $cover.url -}} {{- end -}} {{ .Title }} {{- if or $date $readingTime -}} {{- with $date -}}{{ . }}{{- end -}} {{- with $readingTime -}}{{ . }}{{- end -}} {{- end -}} {{- with $summary -}} {{- . -}} {{- end -}} {{- end -}} {{- end -}}
