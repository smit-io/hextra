{{- /* KaTeX CSS loader Behavior (driven by site.params.math.katex): - base (remote URL) + optional css: - Construct remote CSS URL: "{{ base }}/{{ css | default "katex[.min].css" }}". - Fetch via resources.GetRemote, rewrite font URLs to "{{ base }}/fonts/...". - Build and fingerprint; emit . - base (local path or not set) + css (asset path): - Read CSS from Hugo assets via resources.Get; DO NOT rewrite font URLs. - Build and fingerprint; emit . - base (local path) only (no css): - Link directly to "{{ base }}/katex[.min].css" (no processing). - Nothing set: - Default to a KaTeX release line compatible with Hugo; same as remote path above. Additional: - assets: optional list to publish extra assets. Each entry is an asset path, a glob containing "*", or a directory path ending in "/" (published recursively). CSS/JS get tags with integrity (JS loads async). */ -}} {{- $noop := .WordCount -}} {{- $katexBase := "" -}} {{- with site.Params.math.katex.base -}} {{- $katexBase = . -}} {{- else -}} {{- if not site.Params.math.katex.css -}} {{- /* KaTeX 0.18 introduces breaking CSS class changes, so the stylesheet must match Hugo's embedded KaTeX renderer: 0.16 before Hugo 0.166.0, and 0.18.4 starting with Hugo 0.166.0. See https://github.com/gohugoio/hugo/pull/15259. */ -}} {{- $katexVersion := "0.16" -}} {{- if ge hugo.Version "0.166.0" -}} {{- $katexVersion = "0.18.4" -}} {{- end -}} {{- $katexBase = printf "https://cdn.jsdelivr.net/npm/katex@%s/dist" $katexVersion -}} {{- end -}} {{- end -}} {{- $katexCssAsset := "" -}} {{- with site.Params.math.katex.css -}} {{- $katexCssAsset = . -}} {{- end -}} {{- $s := newScratch -}} {{- $isRemoteBase := or (strings.HasPrefix $katexBase "http://") (strings.HasPrefix $katexBase "https://") -}} {{- /* CSS retrieval consolidated: get raw CSS from either local asset or remote, then process once */ -}} {{- $minSuffix := cond hugo.IsProduction ".min" "" -}} {{- if $isRemoteBase -}} {{- $cssPath := cond (ne $katexCssAsset "") $katexCssAsset (printf "katex%s.css" $minSuffix) -}} {{- $katexCssUrl := urls.JoinPath $katexBase $cssPath -}} {{- with try (resources.GetRemote $katexCssUrl) -}} {{- with .Err -}} {{- errorf "Could not retrieve KaTeX css file from %s. Reason: %s." $katexCssUrl . -}} {{- else with .Value -}} {{- $s.Set "katexCssValue" .Content -}} {{- end -}} {{- end -}} {{- else if $katexCssAsset -}} {{- with resources.Get $katexCssAsset -}} {{- $s.Set "katexCssValue" .Content -}} {{- else -}} {{- errorf "KaTeX css asset not found at %q" $katexCssAsset -}} {{- end -}} {{- end -}} {{- with $s.Get "katexCssValue" -}} {{- $cssContent := . -}} {{- if $isRemoteBase -}} {{- $fontPattern := "url(fonts/" -}} {{- $fontSub := printf "url(%s/" (urls.JoinPath $katexBase "fonts") -}} {{- $cssContent = strings.Replace $cssContent $fontPattern $fontSub -}} {{- end -}} {{- with resources.FromString (printf "css/katex%s.css" $minSuffix) $cssContent -}} {{- $css := . | fingerprint "sha512" -}} {{- end -}} {{- else -}} {{- if not $isRemoteBase -}} {{- $cssPath := cond (ne $katexCssAsset "") $katexCssAsset (printf "katex%s.css" $minSuffix) -}} {{- end -}} {{- end -}} {{- /* Optionally publish files (fonts, css, js, etc.) from assets and emit tags for css/js with integrity and crossorigin. An entry ending in "/" publishes that asset directory recursively; an entry containing "*" is treated as a glob. */ -}} {{- with site.Params.math.katex.assets -}} {{- range $entry := . -}} {{- $files := slice -}} {{- if strings.HasSuffix $entry "/" -}} {{- $files = resources.Match (printf "%s**" $entry) -}} {{- else if strings.Contains $entry "*" -}} {{- $files = resources.Match $entry -}} {{- else -}} {{- with resources.Get $entry -}} {{- $files = slice . -}} {{- else -}} {{- errorf "KaTeX asset not found at %q" $entry -}} {{- end -}} {{- end -}} {{- if not $files -}} {{- warnf "KaTeX assets entry %q matched no files." $entry -}} {{- end -}} {{- range $files -}} {{- $name := .Name | lower -}} {{- if strings.HasSuffix $name ".css" -}} {{- $built := . | fingerprint "sha512" -}} {{- else if or (strings.HasSuffix $name ".js") (strings.HasSuffix $name ".mjs") -}} {{- $built := . | fingerprint "sha512" -}} {{- else -}} {{- .Publish -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
