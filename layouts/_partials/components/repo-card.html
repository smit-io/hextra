{{- /* Shared card for the repository shortcodes (github, gitlab, gitea, codeberg, forgejo and friends). Each provider shortcode fetches its own API, normalises the response into the dict below, and calls this partial. Anything the provider cannot supply is simply omitted. @param {string} url Required. Link to the repository. @param {string} title Repository name, e.g. "owner/repo". @param {string} description Short description. @param {string} language Primary language. @param {int} stars Star count. @param {int} forks Fork count. @param {slice} metrics Alternative to stars/forks for providers that count something else, as dicts of `icon`, `label` and `value`. @param {string} icon Provider icon name. @param {string} thumbnail Optional image URL shown above the card body. @param {string} thumbnailFit How the thumbnail fills its box: cover (default) or contain. @example {{ partial "components/repo-card.html" (dict "url" $url "title" $name) }} */ -}} {{- $url := .url -}} {{- $title := .title | default $url -}} {{- $description := .description -}} {{- $language := .language -}} {{- $stars := .stars -}} {{- $forks := .forks -}} {{- $icon := .icon -}} {{- $thumbnail := .thumbnail -}} {{- /* Stars and forks are the common case, so they stay first-class parameters; providers that count something else (likes, downloads) pass `metrics`. */ -}} {{- $metrics := .metrics | default slice -}} {{- if not $metrics -}} {{- with $stars -}} {{- $metrics = $metrics | append (dict "icon" "star" "label" ((T "stars") | default "stars") "value" .) -}} {{- end -}} {{- with $forks -}} {{- $metrics = $metrics | append (dict "icon" "share" "label" ((T "forks") | default "forks") "value" .) -}} {{- end -}} {{- end -}} {{- with $thumbnail -}} {{- end -}} {{- with $icon -}} {{- partial "utils/icon" (dict "name" . "attributes" `height=16 aria-hidden="true"`) -}} {{- end -}} {{ $title }} {{- with $description -}} {{ . }} {{- end -}} {{- if or $language $metrics -}} {{- with $language -}} {{ . }} {{- end -}} {{- range $metrics -}} {{- partial "utils/icon" (dict "name" .icon "attributes" `height=16 aria-hidden="true"`) -}} {{- partial "inline/compact-number.html" .value }} {{ .label }} {{- end -}} {{- end -}} {{- /* 2889 reads better as 2.9k in a card this size. */ -}} {{- define "_partials/inline/compact-number.html" -}} {{- $n := . | int -}} {{- $out := printf "%d" $n -}} {{- if ge $n 1000000 -}} {{- $out = printf "%.1fM" (div (float $n) 1000000.0) -}} {{- else if ge $n 1000 -}} {{- $out = printf "%.1fk" (div (float $n) 1000.0) -}} {{- end -}} {{- $out = replaceRE `\.0([kM])$` "$1" $out -}} {{- return $out -}} {{- end -}}
