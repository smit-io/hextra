{{- /* Render a card for a Hugging Face model or dataset, with its task, likes and download count. Ported from the Blowfish theme (MIT, Nuno Coração). @param {string} model Model as "owner/name", e.g. "google-bert/bert-base-uncased". @param {string} dataset Dataset as "owner/name", e.g. "stanfordnlp/imdb". @param {bool} showThumbnail Show the owner's avatar. Default false. @example {{< huggingface model="google-bert/bert-base-uncased" >}} @example {{< huggingface dataset="stanfordnlp/imdb" >}} */ -}} {{- $model := .Get "model" -}} {{- $dataset := .Get "dataset" -}} {{- $showThumbnail := eq (printf "%v" (.Get "showThumbnail")) "true" -}} {{- if and $model $dataset -}} {{- errorf "The %q shortcode takes either a 'model' or a 'dataset', not both. See %s" .Name .Position -}} {{- end -}} {{- $id := $model | default $dataset -}} {{- if not $id -}} {{- errorf "The %q shortcode requires a 'model' or 'dataset' parameter. See %s" .Name .Position -}} {{- end -}} {{- if not (findRE `^[A-Za-z0-9._-]+(/[A-Za-z0-9._-]+)?$` $id) -}} {{- errorf "The %q shortcode was given an invalid id %q. See %s" .Name $id .Position -}} {{- end -}} {{- $kind := cond (ne $dataset "") "datasets" "models" -}} {{- $path := cond (ne $dataset "") (printf "datasets/%s" $id) $id -}} {{- $card := dict "url" (printf "https://huggingface.co/%s" $path) "title" $id "icon" "huggingface" -}} {{- $enabled := ne (printf "%v" site.Params.repoCards.enable) "false" -}} {{- if $enabled -}} {{- $api := printf "https://huggingface.co/api/%s/%s" $kind $id -}} {{- with try (resources.GetRemote $api) -}} {{- with .Err -}} {{- warnf "huggingface: unable to fetch %q, rendering a plain card: %s" $api . -}} {{- else with .Value -}} {{- with . | transform.Unmarshal -}} {{- /* The task tag is the most useful one-line summary the API offers for a model; datasets have no equivalent, so they fall back to the kind. */ -}} {{- $language := .pipeline_tag | default (cond (eq $kind "datasets") "dataset" "model") -}} {{- /* Hugging Face counts likes and downloads, not stars and forks. */ -}} {{- $metrics := slice -}} {{- with .likes -}} {{- $metrics = $metrics | append (dict "icon" "heart" "label" ((T "likes") | default "likes") "value" .) -}} {{- end -}} {{- with .downloads -}} {{- $metrics = $metrics | append (dict "icon" "download" "label" ((T "downloads") | default "downloads") "value" .) -}} {{- end -}} {{- $card = merge $card (dict "title" (.id | default $card.title) "language" $language "metrics" $metrics ) -}} {{- with .cardData -}} {{- with .description -}} {{- $card = merge $card (dict "description" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Self-hosted rather than linked, so the card makes no third-party request. */ -}} {{- if and $showThumbnail (in $id "/") -}} {{- $owner := index (split $id "/") 0 -}} {{- with try (resources.GetRemote (printf "https://huggingface.co/api/organizations/%s/avatar" $owner)) -}} {{- with .Err -}} {{- warnf "huggingface: unable to fetch the avatar for %q: %s" $owner . -}} {{- else with .Value -}} {{- with resources.Copy (printf "images/huggingface/%s.png" $owner) . -}} {{- $card = merge $card (dict "thumbnail" .RelPermalink "thumbnailFit" "contain") -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- partial "components/repo-card.html" $card -}}
