{{- /* Embed a GitHub gist. The gist is fetched from the GitHub API at build time and rendered with the theme's own code block, so it gets Chroma highlighting, the filename header and the copy button, and the page loads no third-party script. Ported from the Blowfish theme (MIT, Nuno Coração), which embeds GitHub's gist.js. That script uses document.write, ships GitHub's own stylesheet, and renders nothing when scripting is off. @param {string} id Required. The gist id. @param {string} user The gist owner. Accepted for compatibility; the API does not need it. @param {string} file Render only this file from the gist. @param {string} hl_lines Lines to highlight, e.g. "34-38 43". A gist carries no highlighting metadata of its own, so this is given at the call site. @param {bool} lineNos Show line numbers. @param {int} lineNoStart First line number. Default 1. or @param {string} 0 The gist owner. @param {string} 1 The gist id. @param {string} 2 Optional filename. @example {{< gist id="6cad326836d38bd3a7ae" >}} @example {{< gist "octocat" "6cad326836d38bd3a7ae" "hello_world.rb" >}} */ -}} {{- $id := .Get "id" -}} {{- $file := .Get "file" -}} {{- if not $id -}} {{- $id = .Get 1 -}} {{- $file = .Get 2 -}} {{- end -}} {{- if not $id -}} {{- errorf "The %q shortcode requires a gist id. See %s" .Name .Position -}} {{- end -}} {{- if not (findRE `^[0-9a-f]+$` $id) -}} {{- errorf "The %q shortcode was given %q, which is not a gist id. See %s" .Name $id .Position -}} {{- end -}} {{- $api := printf "https://api.github.com/gists/%s" $id -}} {{- $gist := false -}} {{- /* Skipped entirely for a build that must not reach the network. */ -}} {{- if partial "utils/remote-fetch-enabled.html" . -}} {{- with try (resources.GetRemote $api (dict "headers" (dict "Accept" "application/vnd.github+json"))) -}} {{- with .Err -}} {{- warnf "gist: unable to fetch %q: %s" $api . -}} {{- else with .Value -}} {{- $gist = . | transform.Unmarshal -}} {{- else -}} {{- warnf "gist: %q returned no content" $api -}} {{- end -}} {{- end -}} {{- end -}} {{- $copyEnabled := or (eq site.Params.highlight.copy.enable nil) site.Params.highlight.copy.enable -}} {{- /* Chroma options. A gist has no highlighting metadata, so it comes from here. */ -}} {{- $options := dict -}} {{- with .Get "hl_lines" -}} {{- $options = merge $options (dict "hl_lines" .) -}} {{- end -}} {{- if eq (printf "%v" (.Get "lineNos")) "true" -}} {{- $options = merge $options (dict "lineNos" true) -}} {{- end -}} {{- with .Get "lineNoStart" -}} {{- $options = merge $options (dict "lineNoStart" (int .)) -}} {{- end -}} {{- with $gist -}} {{- $htmlURL := .html_url | default (printf "https://gist.github.com/%s" $id) -}}
{{- range $name, $meta := .files -}} {{- if or (not $file) (eq $name $file) -}} {{- $lang := lower ($meta.language | default "") -}}
{{- partial "components/codeblock" (dict "filename" $name "base_url" "" "lang" $lang "content" ($meta.content | default "") "options" $options "action" (dict "url" $htmlURL "icon" "github" "label" ((T "viewOnGitHub") | default "View on GitHub") ) ) -}} {{- if $copyEnabled -}} {{- partial "components/codeblock-copy-button" (dict "filename" $name) -}} {{- end -}}
{{- if $meta.truncated -}} {{- warnf "gist: %q is truncated by the GitHub API and is shown in part only" $name -}} {{- end -}} {{- end -}} {{- end -}}
{{- else -}} {{- /* No network, or the gist is gone: leave a link rather than nothing. */ -}}

{{- end -}}
