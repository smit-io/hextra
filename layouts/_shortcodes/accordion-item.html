{{- /* A single collapsible section within the `accordion` shortcode. @param {string} title The section heading. @param {string} icon Optional icon shown before the heading. @param {bool} open Whether the section starts open. Default false. @example {{< accordion-item title="What is included?" icon="check" >}}Content{{< /accordion-item >}} */ -}} {{- $title := .Get "title" | default (.Get 0) -}} {{- $icon := .Get "icon" -}} {{- $open := eq (printf "%v" (.Get "open")) "true" -}} {{- if not $title -}} {{- errorf "The %q shortcode requires a 'title' parameter. See %s" .Name .Position -}} {{- end -}} {{- /* In collapse mode every section in the group shares a name, which is what makes the browser close the others when one opens. The name is scoped to the whole document, so it has to be unique per accordion. The parent's ordinal is not: it restarts inside every container, so an accordion nested in a tab took the same number as the first top-level one and the two ended up in a single exclusive group - opening a section inside the tab collapsed an unrelated section outside it. The parent's source position is unique per accordion, so hash that instead. */ -}} {{- $mode := "multiple" -}} {{- $group := "" -}} {{- with .Parent -}} {{- $mode = .Get "mode" | default "multiple" -}} {{- if eq $mode "collapse" -}} {{- $group = printf "hextra-accordion-%s" (substr (md5 (printf "%v" .Position)) 0 10) -}} {{- end -}} {{- end -}}
{{- with $icon -}} {{- partial "utils/icon" (dict "name" . "attributes" `height=16 aria-hidden="true"`) -}} {{- end -}} {{ $title | markdownify }}
