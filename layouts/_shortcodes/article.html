{{- /* Embed a link to another page as a card, with its cover, meta and summary. Ported from the Blowfish theme (MIT, Nuno Coração). As in the original, a card pointing at its own page renders nothing. @param {string} link Required. Path or permalink of the page, e.g. "/blog/my-post". @param {bool} showSummary Show the page summary. Default true. @param {bool} compactSummary Clamp the summary to a single line. Default false. @param {bool} cover Show the page's cover image when it has one. Default true. @example {{< article link="/blog/my-post" >}} @example {{< article link="/blog/my-post" compactSummary=true cover=false >}} */ -}} {{- $link := .Get "link" -}} {{- if not $link -}} {{- errorf "The %q shortcode requires a 'link' parameter. See %s" .Name .Position -}} {{- end -}} {{- $self := "" -}} {{- with .Page -}} {{- $self = .RelPermalink -}} {{- end -}} {{- $target := "" -}} {{- with $link -}} {{- $target = site.GetPage . -}} {{- end -}} {{- if not $target -}} {{- errorf "The %q shortcode was unable to find %q. See %s" $.Name $link $.Position -}} {{- end -}} {{- with $target -}} {{- if eq .RelPermalink $self -}} {{- warnf "The 'article' shortcode links to its own page, skipping." -}} {{- end -}} {{- end -}} {{- partial "shortcodes/article-card.html" (dict "page" $target "self" $self "showSummary" (.Get "showSummary") "compactSummary" (.Get "compactSummary") "cover" (.Get "cover") ) -}}
