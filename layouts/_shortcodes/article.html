{{- /* Embed a link to another page as a card, with its cover, meta and summary. Ported from the Blowfish theme (MIT, Nuno Coração). As in the original, a card pointing at its own page renders nothing. This shortcode runs while Hugo is already rendering page content (search-data .json renders every page), so it avoids touching another page's .Content and resolves meta inline rather than through partials that expect a live page context. @param {string} link Required. Path or permalink of the page, e.g. "/blog/my-post". @param {bool} showSummary Show the page summary. Default true. @param {bool} compactSummary Clamp the summary to a single line. Default false. @param {bool} cover Show the page's cover image when it has one. Default true. @example {{< article link="/blog/my-post" >}} @example {{< article link="/blog/my-post" compactSummary=true cover=false >}} */ -}} {{- $link := .Get "link" -}} {{- if not $link -}} {{- errorf "The %q shortcode requires a 'link' parameter. See %s" .Name .Position -}} {{- end -}} {{- /* Booleans are compared through printf so an explicit `false` is honoured: `default` treats false as empty and would silently restore the default. */ -}} {{- $showSummary := ne (printf "%v" (.Get "showSummary")) "false" -}} {{- $compactSummary := eq (printf "%v" (.Get "compactSummary")) "true" -}} {{- $showCover := ne (printf "%v" (.Get "cover")) "false" -}} {{- $self := "" -}} {{- with .Page -}} {{- $self = .RelPermalink -}} {{- end -}} {{- $target := "" -}} {{- with $link -}} {{- $target = site.GetPage . -}} {{- end -}} {{- if not $target -}} {{- errorf "The %q shortcode was unable to find %q. See %s" $.Name $link $.Position -}} {{- end -}} {{- with $target -}} {{- if eq .RelPermalink $self -}} {{- warnf "The 'article' shortcode links to its own page, skipping." -}} {{- else -}} {{- $cover := dict -}} {{- if $showCover -}} {{- $cover = partial "utils/page-image.html" (dict "page" . "process" "Resize 400x webp q80") -}} {{- end -}} {{- $date := "" -}} {{- if not .Date.IsZero -}} {{- $date = .Date | time.Format (site.Params.dateFormat | default ":date_long") -}} {{- end -}} {{- $readingTime := "" -}} {{- if gt .ReadingTime 0 -}} {{- $readingTime = printf ((T "readingTime") | default "%d min read") .ReadingTime -}} {{- end -}} {{- $summary := "" -}} {{- if $showSummary -}} {{- $summary = .Description | plainify | htmlUnescape -}} {{- if not $summary -}} {{- $summary = .Summary | plainify | htmlUnescape | chomp -}} {{- end -}} {{- end -}} {{- with $cover.url -}} {{- end -}} {{ .Title }} {{- if or $date $readingTime -}} {{- with $date -}}{{ . }}{{- end -}} {{- with $readingTime -}}{{ . }}{{- end -}} {{- end -}} {{- with $summary -}} {{- . -}} {{- end -}} {{- end -}} {{- end -}}
