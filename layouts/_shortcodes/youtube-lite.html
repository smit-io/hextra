{{- /* Embed a YouTube video as a facade: a poster image and a play button that swap themselves for the real player on click. Hugo's built-in `youtube` shortcode loads YouTube's player iframe with the page, costing several hundred kilobytes and setting third-party cookies before anyone presses play. This loads an image instead, and only contacts YouTube once the reader asks for the video. The poster is fetched at build time and served from this site, so a page with an embed makes no third-party request at all until it is played. If the fetch fails the poster falls back to YouTube's CDN. Ported from the Blowfish theme (MIT, Nuno Coração), which wraps the lite-youtube-embed library; this implementation has no dependency. @param {string} id Required. The YouTube video id. @param {string} label Accessible name for the play button, e.g. the video title. @param {string} params Extra player parameters, e.g. "start=130&controls=0". @example {{< youtube-lite id="dQw4w9WgXcQ" label="Never Gonna Give You Up" >}} */ -}} {{- $id := .Get "id" | default (.Get 0) -}} {{- if not $id -}} {{- errorf "The %q shortcode requires an 'id' parameter. See %s" .Name .Position -}} {{- end -}} {{- if not (findRE `^[A-Za-z0-9_-]{11}$` $id) -}} {{- errorf "The %q shortcode was given %q, which is not a YouTube video id. See %s" .Name $id .Position -}} {{- end -}} {{- $params := .Get "params" | default "" -}} {{- if and $params (not (findRE `^[A-Za-z0-9_=&.,%-]+$` $params)) -}} {{- warnf "youtube-lite: dropping unsafe params %q. See %s" $params .Position -}} {{- $params = "" -}} {{- end -}} {{- /* default "" normalises the missing case: with positional parameters .Get on a string key returns nil, `ne nil ""` is true, and the button rendered aria-label="Play video: %!s()". The activation script copies that string into the iframe's title, so a screen-reader user heard the format error both before and after playback. */ -}} {{- $label := .Get "label" | default "" -}} {{- $buttonLabel := cond (ne $label "") (printf "%s: %s" ((T "playVideo") | default "Play video") $label) ((T "playVideo") | default "Play video") -}} {{- /* Self-host the poster so an unplayed embed makes no third-party request. */ -}} {{- $remotePoster := printf "https://i.ytimg.com/vi/%s/hqdefault.jpg" $id -}} {{- $poster := $remotePoster -}} {{- with try (resources.GetRemote $remotePoster) -}} {{- with .Err -}} {{- warnf "youtube-lite: unable to fetch the poster for %q, falling back to YouTube's CDN: %s" $id . -}} {{- else with .Value -}} {{- with resources.Copy (printf "images/youtube/%s.jpg" $id) . -}} {{- $poster = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- .Page.Store.Set "hasYoutubeLite" true -}}
